        """

        pred/target shape: (batch, 6) [min_low_20, max_high_20, mean_close_20, min_low_50, max_high_50, mean_close_50]
        """
        mse = nn.MSELoss(reduction='none')
        weights = torch.tensor([1.0, 1.0, 0.75, 0.75, 0.5, 0.5, 0.25, 0.25],
                               device=pred.device)  # 10-step: 1.0,  20-step: .75, 50-step: 0.5
        loss = mse(pred, target) * weights
        return loss.mean()

        # CNN Feature Extraction
        x = x.transpose(1, 2)  # (batch, features, seq_len)
        x = torch.relu(self.bn1(self.conv1(x)))
        x = self.pool1(x)
        x = torch.relu(self.bn2(self.conv2(x)))
        x = self.pool2(x)
        # LSTM Processing
        x = x.transpose(1, 2)  # (batch, seq_len, features)
        x, _ = self.lstm1(x)  # (batch, seq_len, 512)
        x, _ = self.lstm2(x)  # (batch, seq_len, 256)
        # Attention Mechanism
        attn_output, _ = self.attention(x, x, x)  # Self-attention: (batch, seq_len, 256)
        x = attn_output[:, -1, :]  # Last time step: (batch, 256)
        # Fully Connected Layers
        x = torch.relu(self.fc1(x))
        x = self.dropout1(x)
        x = torch.relu(self.fc2(x))
        x = self.dropout2(x)
        x = self.fc3(x)  # (batch, num_outputs)
        return x
